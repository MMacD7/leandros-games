async function poll(page, fn, ms = 90000) {
  const t0 = Date.now();
  while (Date.now() - t0 < ms) {
    if (await page.evaluate(fn).catch(() => false)) return;
    await page.waitForTimeout(120);
  }
  throw new Error('timed out');
}

      for (let f = 0; f < 160; f++) {
        G.sim(1 / 30);
        const past = Math.hypot(p.pos.x - roof.x, p.pos.z - roof.z) - roof.r;
        if ((s.state === 'swing' || s.state === 'air') && past >= 2 && p.pos.y > roof.h - 8) break;
      }
    });
    await shoot('swing', () => window.__fields());

      for (let f = 0; f < 90; f++) {
        G.sim(1 / 30);
        const above = s.towers.some(t => t !== roof && p.pos.y > t.h + 1);
        if (G.game.world.time - t0 >= 1 && above && s.hooked) break;
      }
    });

      const shout = document.getElementById('shout'); if (shout) { shout.classList.remove('show'); shout.textContent = ''; }

      if (G.game.westies.active) G.game.westies.end();

    await shoot('rb-0.9', () => window.__fields());
